# requirements-cpu.txt
#
# Полный набор зависимостей для ЛОКАЛЬНОЙ машины разработчика (CPU).
# Собран слиянием исходных requirements.txt video-pipeline и
# requirements_full.txt openai-whisper (см. COLAB_MIGRATION_PLAN.md, раздел 3)
# + добавлены пакеты, которые реально импортируются в коде, но отсутствовали
# в обоих исходных файлах: fastapi, uvicorn (services/subtitle_api.py),
# elevenlabs (core/narration/elevenlabs_tts.py), gradio (web/gradio_app.py).
#
# torch/torchaudio/torchvision — CPU-сборки (+cpu). Для Colab/GPU
# используйте requirements-gpu.txt — там эти три пакета исключены и
# ставятся отдельной командой с CUDA-индексом (см. notebooks/colab_setup.ipynb).
#
# ВАЖНО: torch/torchaudio/torchvision с суффиксом +cpu физически не
# резолвятся с обычного PyPI (index.pypi.org не публикует локальные версии
# вида "+cpu") — обязательно нужен --extra-index-url на официальный индекс
# PyTorch, иначе pip install упадёт на первом же пакете с "No matching
# distribution found for torch==...+cpu". Это подтверждено практической
# попыткой установки при подготовке этого файла, не гипотеза.
#
# Установка:
#   python3 -m venv .venv
#   source .venv/bin/activate      (Windows: .venv\Scripts\activate)
#   pip install -r requirements-cpu.txt --extra-index-url https://download.pytorch.org/whl/cpu
#   pip install -e . --no-deps
#   playwright install firefox     # для scraping/ (CamoufoxSession)
#   camoufox fetch                 # докачивает патченный браузер Camoufox
